rows=[]
receipt=json.loads(Path('/tmp/opencode/wl240-build-a/evidence.json').read_text())
def run(name, cmd, env=None):
    start=datetime.datetime.now(datetime.timezone.utc).isoformat()
    r=subprocess.run(cmd,capture_output=True,text=True,env=env,timeout=180)
    (base/'evidence'/('upgrade-'+name+'.txt')).write_text(r.stdout+r.stderr)
    rows.append(dict(test=name,status='PASS' if r.returncode==0 else 'FAIL',exit_code=r.returncode,
                     source_sha=receipt['SOURCE_GIT_SHA'],zip_sha256=receipt['ZIP_SHA256'],
                     started_at=start,ended_at=datetime.datetime.now(datetime.timezone.utc).isoformat(),evidence='upgrade-'+name+'.txt'))
    (base/'evidence/upgrade-results.json').write_text(json.dumps(rows,indent=2)+'\n')
    print(name,r.returncode,flush=True)
    if r.returncode: raise RuntimeError(r.stdout+r.stderr)
