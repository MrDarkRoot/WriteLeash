import hashlib
import json

rows = []
sha = 'ba249dc96d3ea8318df0f3e65cf5a0833422e10a'
zipsha = hashlib.sha256(archive.read_bytes()).hexdigest()
def run(name, cmd, env=None, expected=0):
    result = subprocess.run(cmd, capture_output=True, text=True, env=env, timeout=180)
    (base / 'evidence' / (name + '.txt')).write_text(result.stdout + result.stderr)
    rows.append(dict(test=name, source_sha=sha, zip_sha256=zipsha, status='PASS' if result.returncode == expected else 'FAIL',
                     exit_code=result.returncode, expected_exit=expected, evidence=name + '.txt'))
    (base / 'evidence/setup.json').write_text(json.dumps(rows, indent=2) + '\n')
    print(name, result.returncode, flush=True)
    if result.returncode != expected:
        raise RuntimeError(name + ': ' + result.stdout + result.stderr)
    return result.stdout
